window.addShop = async ()=>{ const name=document.getElementById('shopName').value.trim(); const phone=document.getElementById('shopPhone').value.trim().replace(/\D/g,''); const ownerEmail=document.getElementById('shopOwnerEmail').value.trim().toLowerCase(); const ownerPass=document.getElementById('shopOwnerPass').value; const city=document.getElementById('shopCity').value; const area=document.getElementById('shopArea').value.trim(); const category=document.getElementById('shopCategory').value; const open=document.getElementById('shopOpen').value||'09:00 AM'; const close=document.getElementById('shopClose').value||'09:00 PM'; const insta=document.getElementById('shopInsta').value.trim(); const fb=document.getElementById('shopFb').value.trim(); const map=document.getElementById('shopMap').value.trim(); if(!name||!phone||!city) return alert('Name, Mobile, City required'); if(phone.length!==10) return alert('Mobile 10 digits'); if(!ownerEmail) return alert('Owner Email required - So owner can login at /shop/'); if(!ownerPass || ownerPass.length<6) return alert('Owner Password required min 6 chars'); // 1. Mobile UNIQUE check - Across all shops const qPhone=query(collection(db,"shops"),where("phone","==",phone)); const snapPhone=await getDocs(qPhone); if(!snapPhone.empty) return alert('❌ Mobile '+phone+' already exists - Must be UNIQUE across all shops'); // 2. Email UNIQUE check - NEW - One email per shop const qEmail=query(collection(db,"shops"),where("ownerEmail","==",ownerEmail)); const snapEmail=await getDocs(qEmail); if(!snapEmail.empty) return alert('❌ Owner Email '+ownerEmail+' already exists - One email per shop - Use different email'); setMsg('Creating owner account '+ownerEmail+'...', ''); try{ // Create Firebase Auth user for owner try{ const secondaryApp = initializeApp(firebaseConfig, 'secondary'); const secondaryAuth = getAuth(secondaryApp); await createUserWithEmailAndPassword(secondaryAuth, ownerEmail, ownerPass); setMsg('✅ Owner account created: '+ownerEmail, 'ok'); }catch(e){ if(e.code==='auth/email-already-in-use'){ setMsg('Owner email already exists in Auth - Using existing account', ''); } else { console.log('Owner creation error:', e.code); } } // Create shop with ownerEmail field await addDoc(collection(db,"shops"),{ name,phone,ownerEmail, city,area,category,open,close,insta,fb,map, approved:false, plan:"Rs 500/week per category container", popular:false,banner:false, views:0,callClicks:0,mapClicks:0,bookClicks:0, createdAt:serverTimestamp(), createdBy:currentUser?.email, ownerCreated:true, timingDays:["Mon","Tue","Wed","Thu","Fri","Sat","Sun"] }); alert('✅ Shop added!\n\nOwner Email: '+ownerEmail+'\nOwner Password: '+ownerPass+'\n\nOwner can login at /shop/ with email+password\n\nMobile UNIQUE: '+phone+'\nEmail UNIQUE: '+ownerEmail); ['shopName','shopPhone','shopOwnerEmail','shopOwnerPass','shopArea'].forEach(id=>{ const el=document.getElementById(id); if(el) el.value=''; }); setMsg('✅ Shop added for owner '+ownerEmail, 'ok'); }catch(err){ setMsg('❌ Error: '+err.code+' '+err.message, 'err'); console.error(err); } };
